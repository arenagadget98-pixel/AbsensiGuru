    function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetM = ss.getSheetByName("Anggota") || ss.insertSheet("Anggota");
  var sheetH = ss.getSheetByName("Riwayat") || ss.insertSheet("Riwayat");
  var sheetS = ss.getSheetByName("Pengaturan") || ss.insertSheet("Pengaturan");
  
  var membersData = sheetM.getDataRange().getValues();
  var historyData = sheetH.getDataRange().getValues();
  var settingsData = sheetS.getDataRange().getValues();
  
  var members = [];
  if (membersData.length > 1) {
    for (var i = 1; i < membersData.length; i++) {
      if (membersData[i][0] !== "" && membersData[i][0] !== null) {
        members.push({ 
          id: String(membersData[i][0]).replace(/^'/, ''), 
          name: String(membersData[i][1] || ""), 
          class: String(membersData[i][2] || ""),
          nfcId: String(membersData[i][3] || "").replace(/^'/, '')
        });
      }
    }
  }

  var history = [];
  if (historyData.length > 1) {
    for (var j = 1; j < historyData.length; j++) {
      if (historyData[j][0] !== "" && historyData[j][0] !== null) {
        var rawDate = historyData[j][0];
        var rawTime = historyData[j][1];
        
        if (rawDate instanceof Date) {
          var yyyy = rawDate.getFullYear();
          var mm = ("0" + (rawDate.getMonth() + 1)).slice(-2);
          var dd = ("0" + rawDate.getDate()).slice(-2);
          rawDate = yyyy + "-" + mm + "-" + dd;
        }
        
        if (rawTime instanceof Date) {
          var hh = ("0" + rawTime.getHours()).slice(-2);
          var min = ("0" + rawTime.getMinutes()).slice(-2);
          var sec = ("0" + rawTime.getSeconds()).slice(-2);
          rawTime = hh + ":" + min + ":" + sec;
        }

        history.push({
          date: String(rawDate).replace(/^'/, ''),
          time: String(rawTime).replace(/^'/, ''),
          memberId: String(historyData[j][2] || "").replace(/^'/, ''),
          memberName: String(historyData[j][3] || ""),
          memberClass: String(historyData[j][4] || ""),
          status: String(historyData[j][5] || "")
        });
      }
    }
  }

  var settings = {};
  if (settingsData.length > 1) {
    for (var k = 1; k < settingsData.length; k++) {
      if (settingsData[k][0] !== "" && settingsData[k][0] !== null) {
        var val = settingsData[k][1];
        if (val instanceof Date) {
          var h = ("0" + val.getHours()).slice(-2);
          var m = ("0" + val.getMinutes()).slice(-2);
          val = h + ":" + m;
        } else {
          val = String(val).replace(/^'/, '');
        }
        settings[String(settingsData[k][0])] = val;
      }
    }
  }

  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    members: members,
    history: history,
    settings: settings
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  var hasLock = false;
  try {
    hasLock = lock.tryLock(20000); // Wait up to 20s for lock across devices
  } catch (err) {}

  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. ATOMIC RECORD ATTENDANCE (Multi-Device Safe)
    if (data.action === "recordAttendance") {
      var sheetH = ss.getSheetByName("Riwayat") || ss.insertSheet("Riwayat");
      if (sheetH.getLastRow() === 0) {
        sheetH.appendRow(["Tanggal", "Waktu", "ID", "Nama", "Kelas", "Status"]);
      }

      var existing = sheetH.getDataRange().getValues();
      var targetDate = String(data.record.date).replace(/^'/, '');
      var targetId = String(data.record.memberId).replace(/^'/, '');

      for (var r = 1; r < existing.length; r++) {
        var rowDate = existing[r][0];
        if (rowDate instanceof Date) {
          var yyyy = rowDate.getFullYear();
          var mm = ("0" + (rowDate.getMonth() + 1)).slice(-2);
          var dd = ("0" + rowDate.getDate()).slice(-2);
          rowDate = yyyy + "-" + mm + "-" + dd;
        } else {
          rowDate = String(rowDate).replace(/^'/, '');
        }
        var rowId = String(existing[r][2] || "").replace(/^'/, '');

        if (rowDate === targetDate && rowId === targetId) {
          return ContentService.createTextOutput(JSON.stringify({
            status: "already_present",
            message: "Anggota sudah presensi hari ini"
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }

      sheetH.appendRow([
        "'" + targetDate,
        "'" + String(data.record.time).replace(/^'/, ''),
        "'" + targetId,
        String(data.record.memberName || ""),
        String(data.record.memberClass || ""),
        String(data.record.status || "")
      ]);

      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        message: "Presensi berhasil dicatat"
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // 2. FULL SYNC (FALLBACK & DATA MANAGEMENT)
    if (data.action === "syncAll") {
      var sheetM = ss.getSheetByName("Anggota") || ss.insertSheet("Anggota");
      sheetM.clearContents();
      var rowsM = [["ID", "Nama", "Kelas", "No NFC"]];
      if(data.members && data.members.length > 0) {
        data.members.forEach(function(m) {
          rowsM.push(["'" + String(m.id || ""), String(m.name || ""), String(m.class || ""), "'" + String(m.nfcId || "")]);
        });
      }
      sheetM.getRange(1, 1, rowsM.length, 4).setValues(rowsM);
      
      var sheetH2 = ss.getSheetByName("Riwayat") || ss.insertSheet("Riwayat");
      sheetH2.clearContents();
      var rowsH = [["Tanggal", "Waktu", "ID", "Nama", "Kelas", "Status"]];
      if(data.history && data.history.length > 0) {
        data.history.forEach(function(h) {
          rowsH.push(["'" + String(h.date || ""), "'" + String(h.time || ""), "'" + String(h.memberId || ""), String(h.memberName || ""), String(h.memberClass || ""), String(h.status || "")]);
        });
      }
      sheetH2.getRange(1, 1, rowsH.length, 6).setValues(rowsH);

      if(data.settings) {
        var sheetS = ss.getSheetByName("Pengaturan") || ss.insertSheet("Pengaturan");
        sheetS.clearContents();
        var rowsS = [["Pengaturan", "Nilai"]];
        for (var key in data.settings) {
          rowsS.push([String(key), "'" + String(data.settings[key] || "")]);
        }
        sheetS.getRange(1, 1, rowsS.length, 2).setValues(rowsS);
      }
      return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
    }

    return ContentService.createTextOutput("Unknown Action").setMimeType(ContentService.MimeType.TEXT);

  } catch(err) {
    return ContentService.createTextOutput("Error: " + err.message).setMimeType(ContentService.MimeType.TEXT);
  } finally {
    if (hasLock) {
      lock.releaseLock();
    }
  }
}
